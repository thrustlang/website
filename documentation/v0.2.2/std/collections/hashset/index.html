std::collections::hashset (v0.2.2)

Hash set of keys.

Latest

Overview

std::collections::hashset stores a set of unique keys with insert, contains, and remove.

insert returns true when the key was new; dropHashSet releases the storage.

Behavior and Use

insert returns true when the key was new; contains reports presence; remove deletes a key.

Call dropHashSet when the set is no longer needed to release the backing storage.

Examples

import std::collections::hashset;
import std::io;

// Keep track of which feature flags are enabled.
fn main() s32 @public {
    var enabled := hashset::newHashSet[s32]();

    var flagA: s32 = 7;

    hashset::insert[s32](ref enabled, ref flagA);

    if hashset::contains[s32](ref enabled, ref flagA) {
        io::print("flag A is enabled\n");
    }

    hashset::dropHashSet[s32](ref enabled);

    return 0;
}
import std::collections::hashset;
import std::io;

// Revoke a guest tag from the allowed set and report what remains.
fn main() s32 @public {
    var tags := hashset::newHashSet[s32]();

    var admin: s32 = 1;
    var guest: s32 = 2;

    hashset::insert[s32](ref tags, ref admin);
    hashset::insert[s32](ref tags, ref guest);

    hashset::remove[s32](ref tags, ref guest);

    io::print("remaining tags: %d\n", hashset::getLength[s32](ref tags));

    hashset::dropHashSet[s32](ref tags);

    return 0;
}
import std::collections::hashset;
import std::io;

// Pre-size a set of known VIP ids, then check membership.
fn main() s32 @public {
    var vips := hashset::withCapacity[s32](8);

    var id: s32 = 5;

    hashset::insert[s32](ref vips, ref id);

    if hashset::contains[s32](ref vips, ref id) {
        io::print("welcome, VIP\n");
    }

    hashset::dropHashSet[s32](ref vips);

    return 0;
}

Notes

  • contains reports presence without removing the key.
  • insert returns true only for newly added keys.
